1use std::sync::{Mutex, OnceLock};
2use std::time::{Duration, Instant};
3
4struct ScriptState {
5    last_cycle_start: Instant,
6    jumping: bool,
7}
8
9static STATE: OnceLock<Mutex<ScriptState>> = OnceLock::new();
10
11pub fn main_obj_script() -> (i32, i32) {
12    // this is just an example behavior for main object
13    // if you run 'cargo run --bin main' you'll see the ship "falling" every 3 seconds
14    // so in this funciton you can write custom main object script
15
16    const COOLDOWN: Duration = Duration::from_secs(3);
17    const JUMP_DURATION: Duration = Duration::from_millis(200);
18
19    let state = STATE.get_or_init(|| {
20        Mutex::new(ScriptState {
21            last_cycle_start: Instant::now(),
22            jumping: false,
23        })
24    });
25
26    let mut st = state.lock().unwrap();
27    let elapsed = st.last_cycle_start.elapsed();
28
29    if st.jumping {
30        // currently in jumping phase
31        if elapsed >= JUMP_DURATION {
32            // jump finished
33            st.jumping = false;
34            st.last_cycle_start = Instant::now(); // restart cycle after jump ends
35            return (0, 0);
36        } else {
37            // still jumping
38            return (0, -4); // height of jump
39        }
40    } else {
41        // waiting for the 3-second cooldown
42        if elapsed >= COOLDOWN {
43            // start a new jump
44            st.jumping = true;
45            st.last_cycle_start = Instant::now(); // mark start of jump
46            return (0, -4);
47        } else {
48            // still cooling down
49            return (0, 0);
50        }
51    }
52}
